727 static std::vector<int> parse_ints(std::string const &number_string)
728 {
729 std::stringstream number_stream(number_string);
730 std::vector<int> result;
731 result.reserve(max_num_dimensions);
732 while (!number_stream.eof())
733 {
734 std::string word;
735 number_stream >> word;
736 int temp_int = -1;
737
738 // remove any leading or trailing '"'
739 size_t pos = word.find_first_of('\"');
740 if (pos != std::string::npos)
741 word.erase(word.begin() + pos);
742
743 if (std::stringstream(word) >> temp_int)
744 {
745 if (result.size() == max_num_dimensions)
746 { // too many ints, return invalid result
747 result.clear();
748 return result;
749 }
750 else
751 result.push_back(temp_int);
752 }
753 }
754
755 return result;
756 }
769 template<typename out_type>
770 std::optional<out_type> get_val(std::vector<std::string> const &strs,
771 std::string_view const &s) const
772 {
773 static_assert(std::is_same_v<out_type, int> or std::is_same_v<out_type, bool>
774 or std::is_same_v<out_type, float> or std::is_same_v<out_type, double>
775 or std::is_same_v<out_type, std::string>,
776 "prog_opts can only process: int, float, double, bool or string");
777 for (size_t i = 0; i < strs.size(); i++)
778 {
779 if (strs[i] == s)
780 {
781 try {
782 if (i + 1 == strs.size())
783 return {};
784 const std::string &val = strs[i + 1];
785 if constexpr (std::is_same_v<out_type, std::string>)
786 return val;
787 else if constexpr (std::is_same_v<out_type, int>)
788 return std::stoi(val);
789 else if constexpr (std::is_same_v<out_type, double>)
790 return std::stod(val);
791 else if constexpr (std::is_same_v<out_type, float>)
792 return std::stof(val);
793 else if constexpr (std::is_same_v<out_type, bool>)
794 {
795 if (val == "on" or val == "yes" or val == "enable" or val == "true"
796 or val == "1")
797 return true;
798 else if (val == "off" or val == "no" or val == "disable"
799 or val == "false" or val == "0")
800 return false;
801 else
802 throw std::invalid_argument("bad value for bool");
803 }
804 } catch (std::invalid_argument &) {
805 std::string msg = std::string("invalid value for '") + std::string(s)
806 + "', cannot convert '" + strs[i + 1] + "' to '";
807 if constexpr (std::is_same_v<out_type, std::string>)
808 msg += "string'";
809 else if constexpr (std::is_same_v<out_type, int>)
810 msg += "int'";
811 else if constexpr (std::is_same_v<out_type, double>)
812 msg += "double'";
813 else if constexpr (std::is_same_v<out_type, float>)
814 msg += "float'";
815 else if constexpr (std::is_same_v<out_type, bool>)
816 msg += "bool'";
817
818 throw std::runtime_error(msg);
819 } catch (std::out_of_range &) {
820 std::string msg = std::string("invalid value for '") + std::string(s)
821 + "', out-of-range error in conversion of '" + strs[i + 1] + "' to '";
822 if constexpr (std::is_same_v<out_type, std::string>)
823 msg += "string'";
824 else if constexpr (std::is_same_v<out_type, int>)
825 msg += "int'";
826 else if constexpr (std::is_same_v<out_type, double>)
827 msg += "double'";
828 else if constexpr (std::is_same_v<out_type, float>)
829 msg += "float'";
830 else if constexpr (std::is_same_v<out_type, bool>)
831 msg += "bool'";
832
833 throw std::runtime_error(msg);
834 }
835 }
836 }
837 return {};
838 }
