706 static std::vector<int> parse_ints(std::string const &number_string)
707 {
708 std::stringstream number_stream(number_string);
709 std::vector<int> result;
710 result.reserve(max_num_dimensions);
711 while (!number_stream.eof())
712 {
713 std::string word;
714 number_stream >> word;
715 int temp_int = -1;
716
717 // remove any leading or trailing '"'
718 size_t pos = word.find_first_of('\"');
719 if (pos != std::string::npos)
720 word.erase(word.begin() + pos);
721
722 if (std::stringstream(word) >> temp_int)
723 {
724 if (result.size() == max_num_dimensions)
725 { // too many ints, return invalid result
726 result.clear();
727 return result;
728 }
729 else
730 result.push_back(temp_int);
731 }
732 }
733
734 return result;
735 }
748 template<typename out_type>
749 std::optional<out_type> get_val(std::vector<std::string> const &strs,
750 std::string_view const &s) const
751 {
752 static_assert(std::is_same_v<out_type, int> or std::is_same_v<out_type, bool>
753 or std::is_same_v<out_type, float> or std::is_same_v<out_type, double>
754 or std::is_same_v<out_type, std::string>,
755 "prog_opts can only process: int, float, double, bool or string");
756 for (size_t i = 0; i < strs.size(); i++)
757 {
758 if (strs[i] == s)
759 {
760 try {
761 if (i + 1 == strs.size())
762 return {};
763 const std::string &val = strs[i + 1];
764 if constexpr (std::is_same_v<out_type, std::string>)
765 return val;
766 else if constexpr (std::is_same_v<out_type, int>)
767 return std::stoi(val);
768 else if constexpr (std::is_same_v<out_type, double>)
769 return std::stod(val);
770 else if constexpr (std::is_same_v<out_type, float>)
771 return std::stof(val);
772 else if constexpr (std::is_same_v<out_type, bool>)
773 {
774 if (val == "on" or val == "yes" or val == "enable" or val == "true"
775 or val == "1")
776 return true;
777 else if (val == "off" or val == "no" or val == "disable"
778 or val == "false" or val == "0")
779 return false;
780 else
781 throw std::invalid_argument("bad value for bool");
782 }
783 } catch (std::invalid_argument &) {
784 std::string msg = std::string("invalid value for '") + std::string(s)
785 + "', cannot convert '" + strs[i + 1] + "' to '";
786 if constexpr (std::is_same_v<out_type, std::string>)
787 msg += "string'";
788 else if constexpr (std::is_same_v<out_type, int>)
789 msg += "int'";
790 else if constexpr (std::is_same_v<out_type, double>)
791 msg += "double'";
792 else if constexpr (std::is_same_v<out_type, float>)
793 msg += "float'";
794 else if constexpr (std::is_same_v<out_type, bool>)
795 msg += "bool'";
796
797 throw std::runtime_error(msg);
798 } catch (std::out_of_range &) {
799 std::string msg = std::string("invalid value for '") + std::string(s)
800 + "', out-of-range error in conversion of '" + strs[i + 1] + "' to '";
801 if constexpr (std::is_same_v<out_type, std::string>)
802 msg += "string'";
803 else if constexpr (std::is_same_v<out_type, int>)
804 msg += "int'";
805 else if constexpr (std::is_same_v<out_type, double>)
806 msg += "double'";
807 else if constexpr (std::is_same_v<out_type, float>)
808 msg += "float'";
809 else if constexpr (std::is_same_v<out_type, bool>)
810 msg += "bool'";
811
812 throw std::runtime_error(msg);
813 }
814 }
815 }
816 return {};
817 }
