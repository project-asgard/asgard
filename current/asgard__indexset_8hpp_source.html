237 int64_t find(const int *idx) const
238 {
239 int64_t first = 0, last = num_indexes_ - 1;
240 int64_t current = (first + last) / 2;
241 while (first <= last)
242 {
243 match cmp = compare(current, idx);
244 if (cmp == before_current)
245 {
246 last = current - 1;
247 }
248 else if (cmp == after_current)
249 {
250 first = current + 1;
251 }
252 else // match_found
253 {
254 return current;
255 }
256 current = (first + last) / 2;
257 }
258 return -1;
259 }
261 int find(std::array<int, max_num_dimensions> const &idx) const { return find(idx.data()); }
263 bool missing(const int *idx) const { return (find(idx) == -1); }
265 bool missing(std::array<int, max_num_dimensions> const &idx) const { return missing(idx.data()); }
266
268 indexset &operator+=(indexset const &iset)
269 {
270 expect(iset.num_dimensions_ == num_dimensions_);
271 if (iset.num_indexes_ == 0)
272 return *this;
273
274 std::vector<int> union_set;
275 union_set.reserve(iset.indexes_.size() + indexes_.size());
276
277 int64_t ia = 0;
278 int64_t ib = 0;
279 while (ia < num_indexes_ and ib < iset.num_indexes_)
280 {
281 match cmp = compare(ia, iset[ib]);
282 if (cmp == match::before_current)
283 {
284 union_set.insert(union_set.end(), iset[ib], iset[ib] + num_dimensions_);
285 ib++;
286 }
287 else if (cmp == match::after_current)
288 {
289 union_set.insert(union_set.end(), index(ia),
290 index(ia) + num_dimensions_);
291 ia++;
292 }
293 else
294 {
295 union_set.insert(union_set.end(), index(ia),
296 index(ia) + num_dimensions_);
297 ia++;
298 ib++;
299 }
300 }
301 if (ia < num_indexes_)
302 union_set.insert(union_set.end(), (*this)[ia],
303 (*this)[ia] + (num_indexes_ - ia) * num_dimensions_);
304 else if (ib < iset.num_indexes_)
305 union_set.insert(union_set.end(), iset[ib],
306 iset[ib] + (iset.num_indexes_ - ib) * num_dimensions_);
307
308 indexes_ = std::move(union_set);
309 num_indexes_ = static_cast<int64_t>(indexes_.size() / num_dimensions_);
310 return *this;
311 }
