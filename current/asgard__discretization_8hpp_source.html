488 void progress_report(std::ostream &os = std::cout, int64_t ndof = -1) const {
489 if (stepper.is_steady_state())
490 {
491 os << "refinement iteration " << std::setw(10) << tools::split_style(stepper.data.step());
492 }
493 else
494 {
495 os << "time-step: " << std::setw(10) << tools::split_style(stepper.data.step()) << " time: ";
496 std::string s = std::to_string(stepper.data.time());
497
498 if (s.size() < 7)
499 os << std::setw(10) << s << std::string(7 - s.size(), ' ');
500 else
501 os << std::setw(10) << s;
502 }
503 if (ndof >= 0) {
504 os << " dof: " << std::setw(14) << tools::split_style(ndof);
505 } else {
506 os << " dof: " << std::setw(14) << tools::split_style(terms.grid.num_dof());
507 }
508
509 #ifdef ASGARD_USE_FLOPCOUNTER
510 int64_t const flops = tools::timer.max_flops();
511 if (flops > 0)
512 os << " maxGflops: " << std::to_string(1.E-9 * static_cast<double>(flops));
513 #endif
514
515 int64_t const num_appy = stepper.solver_iterations();
516 if (num_appy > 0) { // using iterative solver
517 os << " av-iter: " << std::setw(14) << tools::split_style(num_appy / stepper.data.step())
518 << '\n';
519 } else {
520 os << '\n';
521 }
522 }

738 reconstruct_solution get_local_snapshot() const
739 {
740 reconstruct_solution shot(
741 num_dims(), terms.grid.num_indexes(), terms.grid[0], degree(), state.data());
742
743 std::array<double, max_num_dimensions> xmin, xmax;
744 for (int d : iindexof(num_dims())) {
745 xmin[d] = domain_.xleft(d);
746 xmax[d] = domain_.xright(d);
747 }
748
749 shot.set_domain_bounds(xmin.data(), xmax.data());
750
751 return shot;
752 }
754 void update_grid(std::vector<precision> &next)
755 {
756 if (is_leader())
757 terms.grid.remap(terms.block_size(), next);
758 terms.prapare_kron_workspace();
759 std::visit([&](auto &p)
760 {
761 if constexpr (std::is_same_v<std::decay_t<decltype(p)>, poisson_1d<precision>>) {
762 p.update_level(terms.grid.current_level(0));
763 } else if constexpr (std::is_same_v<std::decay_t<decltype(p)>, poisson_md<precision>>) {
764 terms.moms.update_position_grid(terms.grid);
765 p.update_preconditioner(terms.moms.get_position_grid(), terms.conn, poisson_bc::periodic);
766 }
767 }, get_poisson());
768 }

771 {
772 if (not is_leader())
773 return;
774 refinement.refine(f, mode, terms);
775 }
776 #ifdef ASGARD_USE_GPU
778 void update_grid(gpu::vector<precision> &next)
779 {
780 if (is_leader())
781 terms.grid.remap(terms.block_size(), next);
782 terms.prapare_kron_workspace();
783 std::visit([&](auto &p)
784 {
