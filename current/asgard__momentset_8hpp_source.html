30struct moment
31{
33 static constexpr moment zero(int num_velocity) {
34 moment m(0);
35 for (int d = 1; d < num_velocity; d++) m.pows[d] = 0;
36 return m;
37 }
40 static constexpr moment electric(dimension_id dim, int num_pos_dims) {
41 moment m = moment::zero(num_pos_dims);
42 m.pows[dim()] = electric_flag;
43 return m;
44 }
46 constexpr moment() : pows{unset_flag, unset_flag, unset_flag} { static_assert(max_mom_dims == 3); }
48 constexpr moment(int pv1) : pows{pv1, unset_flag, unset_flag} { static_assert(max_mom_dims == 3); }
50 constexpr moment(int pv1, int pv2) : pows{pv1, pv2, unset_flag} { static_assert(max_mom_dims == 3); }
52 constexpr moment(int pv1, int pv2, int pv3) : pows{pv1, pv2, pv3} { static_assert(max_mom_dims == 3); }
54 constexpr int num_dims() const {
55 for (int i = 0; i < max_mom_dims; i++)
56 if (pows[i] == unset_flag) return i;
57 return max_mom_dims;
58 }
60 constexpr int operator [] (int d) const { return pows[d]; }
62 bool operator == (moment const &other) const {
63 return std::equal(pows.begin(), pows.end(), other.pows.begin());
64 }
66 bool operator != (moment const &other) const {
67 return not (*this == other);
68 }
70 constexpr bool is_zero() const {
71 for (int const pow : pows) {
72 if (pow == unset_flag) return true;
73 if (pow != 0) return false;
74 }
75 return true;
76 }
78 constexpr bool is_electric() const {
79 for (int const pow : pows)
80 if (pow == electric_flag) return true;
81 return false;
82 }
84 int get_electric_direction() const {
85 for (int const i : iindexof(pows))
86 if (pows[i] == electric_flag) return i;
87 throw std::runtime_error("The moment must be an electric field moment to get its direction");
88 return -1; // unreachable due to rassert
89 }
91 std::string to_string() const {
92 std::string m = (pows[0] == unset_flag) ? "x" : std::to_string(pows[0]);
93 m += (pows[1] == unset_flag) ? "x" : std::to_string(pows[1]);
94 m += (pows[2] == unset_flag) ? "x" : std::to_string(pows[2]);
95 return m;
96 }
98 friend std::ostream& operator<<(std::ostream& os, moment const &m) {
99 if (m.num_dims() == 0) os << "()";
100 else if (m.num_dims() == 1) os << "(" << m.pows[0] << ")";
101 else {
102 os << "(" << m.pows[0];
103 for (int i = 1; i < m.num_dims(); i++)
104 os << ", " << m.pows[i];
105 os << ")";
106 }
107 return os;
108 }
109
111 std::array<int, max_mom_dims> pows;
112
113 private:
114 static constexpr int unset_flag = -1;
115 static constexpr int electric_flag = -0xef; // ef for electric field, this has a decimal value of -239
116};
