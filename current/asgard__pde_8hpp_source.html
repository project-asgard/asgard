768 void check_chain() {
769 if (not std::holds_alternative<std::vector<term_1d<P>>>(extra_))
770 return; // not actually a chain
771 int side = 0;
772 int central = 0;
773 std::vector<term_1d<P>> const &chain_ = std::get<std::vector<term_1d<P>>>(extra_);
774 for (int i : iindexof(chain_)) {
775 if (chain_[i].is_volume())
776 continue;
777 if (chain_[i].flux() == flux_type::central)
778 central ++;
779 else
780 side ++;
781 }
782 rassert(central <= 1, "cannot chain two central fluxes together");
783 rassert(not (central == 1 and side > 0),
784 "cannot chain a central flux with a side flux");
785 rassert(side <= 2, "cannot chain more than two non-central fluxes");
786 }
