424 bool is_valid() const {

468 P eval(P const x[], P t) {
469 std::vector<P> xx(1), fx(1);
470 P v = P{1};
471 for (int d : iindexof(max_num_dimensions)) {
472 v *= std::visit([&](auto const &f) -> P {
473 using current_type = std::decay_t<decltype(f)>;
474 if constexpr (std::is_same_v<current_type, std::monostate>) {
475 return 1; // ignore this dimension
476 } else if constexpr (std::is_same_v<current_type, P>) {
477 return f;
478 } else {
479 xx.front() = x[d];
480 if constexpr (std::is_same_v<current_type, sfixed_func1d<P>>) {
481 f(xx, fx);
482 return fx.front();
483 } else {
484 f(xx, t, fx);
485 return fx.front();
486 }
487 }
488 }, funcs_[d]);
489 }
490 if (is_time_sep()) v *= std::get<scalar_func<P>>(time_func_)(t);
491 return v;
492 }

503 os << "f" << d << "(x)";
504 else

506 if (d + 1 < nd) os << ", ";
507 }
508 os << ")";
