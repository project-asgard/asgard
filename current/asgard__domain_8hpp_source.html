408 bool is_valid() const {
409 int const nd = num_dims();
410 if (is_time_non_sep()) {
411 bool any = false;
412 for (int d : iindexof(nd)) if (is_time_dep(dimension_id{d})) any = true;

414 std::cerr << "Found asgard::separable_func set as non-separable in time "
415 "but without a time-depend spacial component for any dimension. "
416 "This 'smells' of an error, so stopping here.\n";
417 return false;
418 }
419 } else {
420 bool any = false;
421 for (int d : iindexof(nd)) if (is_time_dep(dimension_id{d})) any = true;
422 if (any) {
423 std::cerr << "Found asgard::separable_func set as separable or constant in time "
424 "but some spacial component have a time-dependence. "
425 "This 'smells' of an error, so stopping here.\n";
426 return false;
427 }
428 }
429 for (int d = nd; d < max_num_dimensions; d++) {

452 P eval(P const x[], P t) {
453 std::vector<P> xx(1), fx(1);
454 P v = P{1};
455 for (int d : iindexof(max_num_dimensions)) {
456 v *= std::visit([&](auto const &f) -> P {
457 using current_type = std::decay_t<decltype(f)>;
458 if constexpr (std::is_same_v<current_type, std::monostate>) {
459 return 1; // ignore this dimension
460 } else if constexpr (std::is_same_v<current_type, P>) {
461 return f;
462 } else {
463 xx.front() = x[d];
464 if constexpr (std::is_same_v<current_type, sfixed_func1d<P>>) {
465 f(xx, fx);
466 return fx.front();
467 } else {
468 f(xx, t, fx);
469 return fx.front();
470 }
471 }
472 }, funcs_[d]);
473 }
474 v *= std::visit([&](auto const &f) -> P {
475 using current_type = std::decay_t<decltype(f)>;
476 if constexpr (std::is_same_v<current_type, std::monostate>) {
477 return 1; // ignore this dimension
478 } else if constexpr (std::is_same_v<current_type, P>) {
479 return f;
480 } else {
481 return f(t);
482 }
483 }, time_func_);
484 return v;
485 }

488 void print_stats(std::ostream &os = std::cout) const {
489 int nd = num_dims();
490 os << "separable function: " << nd << "D\n";
491 os << " (";
492 for (int d : iindexof(nd)) {

495 else if (is_fixed(dimension_id{d}))
496 os << "f" << d << "(x)";
497 else

499 if (d + 1 < nd) os << ", ";
500 }
501 os << ")";
